    public sealed class Ruler3000Lines : IDisposable
{
    private CameraDiscovery _discovery;
    private ICamera _camera;
    private FrameGrabber _grabber;
    private IAnyParameters _p;
    private int _width = 0; // X pixel count
    public int WidthPixels => _width;

    public int timeoutMsPerFrame = 200; // GrabNext() timeout (waiting for external trigger)
    // ---- Connection ----
    public void Open(string ctiPath, string cameraIp)
    {
        _discovery = CameraDiscovery.CreateFromProducerFile(ctiPath);
        _camera = _discovery.ConnectTo(IPAddress.Parse(cameraIp));
        _p = _camera.GetAnyParameters();

        // Stop just in case
        TryExec("AcquisitionStop");
    }

    // ---- Pre-acquisition settings (External Trigger / Exposure / Height=1, etc.) ----
    // * Do not call during acquisition
    public void ConfigureForExternalTrigger(double exposureUs = 3000.0, string triggerLine = "Line1")
    {
        TryExec("AcquisitionStop");

        // Parent -> Child (Must is exception, Optional is suppressed)
        SetEnumMust("AcquisitionMode", "Continuous");

        SetEnumMust("TriggerMode", "On");
        TrySetEnum("TriggerSelector", "FrameStart");
        SetEnumMust("TriggerSource", triggerLine);
        TrySetEnum("TriggerActivation", "RisingEdge");

        TrySetEnum("LineSelector", triggerLine);
        TrySetEnum("LineMode", "Input");
        TrySetFloat("LineDebouncerTime", 5.0);

        TrySetEnum("ExposureAuto", "Off");
        SetFloatMust("ExposureTime", exposureUs);
        TrySetEnum("GainAuto", "Off");
        TrySetFloat("Gain", 0.0);

        // The base for line scan is Height=1 (If transferring multiple lines, >1)
        TrySetEnum("PixelFormat", "Mono32f"); // Ignored internally if not supported
        TrySetFloat("Height", 1.0);

        // Determine width
        _width = (int)Math.Round(GetFloatMust("Width"));
        if (_width <= 0) throw new InvalidOperationException("Failed to get Width.");

        _grabber?.Dispose();
        _grabber = _camera.CreateFrameGrabber();
    }

    /// <summary>
    /// Captures until the user-specified lineCount is reached and returns a height map (float) of shape [lineCount, Width].
    /// useTransformed=true: Returns a float conversion of the Z(mm) double source / false: Returns the U16 equivalent value packed into a float.
    /// If you want to save the data, you can call IFrame.Save(basePath) for each frame using onSaveIconPerFrame.
    /// </summary>
    public float[,] CaptureLinesTiled(
        int lineCount,
        int timeoutMsPerFrame = 200,
        bool useTransformed = false,
        Action<int, IFrame> onSaveIconPerFrame = null // e.g.: (i, f) => f.Save($@"C:\cap\frame_{i}")
    )
    {
        if (_grabber == null) throw new InvalidOperationException("Call ConfigureForExternalTrigger() first.");
        if (lineCount <= 0) throw new ArgumentOutOfRangeException(nameof(lineCount));

        // Number of lines included in 1 frame (Height)
        int linesPerFrame = (int)Math.Round(GetFloatMust("Height"));
        if (linesPerFrame <= 0) linesPerFrame = 1;

        // Required number of frames (rounded up) = Same as HALCON's (LineCount + height -1) / height
        int loops = (lineCount + linesPerFrame - 1) / linesPerFrame;

        var hm = new float[lineCount, _width];
        int yWrite = 0;

        _grabber.Start();
        try
        {
            for (int i = 0; i < loops && yWrite < lineCount; i++)
            {
                var res = _grabber.GrabNext(TimeSpan.FromMilliseconds(timeoutMsPerFrame));

                res.IfCompleteFrame(frame =>
                {
                    try
                    {
                        ProcessCompletedFrame(
                            frameIndex: i,
                            frame: frame,
                            useTransformed: useTransformed,
                            hm: hm,
                            yWrite: ref yWrite,
                            lineCount: lineCount,
                            onSaveIconPerFrame: onSaveIconPerFrame
                        );
                    }
                    finally
                    {
                        frame.Release(); // Required for re-queue
                    }
                })
                .IfIncompleteFrame(f => { f.Release(); })
                .IfTimedOut(() => { /* e.g., Trigger not arrived */ });
            }
        }
        finally
        {
            _grabber.Stop();
        }

        return hm;
    }

    /// <summary>
    /// Processes a completed frame from GrabNext() and stacks it into the height map.
    /// ・Saves iCon if necessary (onSaveIconPerFrame)
    /// ・Gets Range and copies it to hm as U16 or double(mm)
    /// </summary>
    private void ProcessCompletedFrame(
        int frameIndex,
        IFrame frame,
        bool useTransformed,
        float[,] hm,
        ref int yWrite,
        int lineCount,
        Action<int, IFrame> onSaveIconPerFrame // nullable
    )
    {
        // 1) Save iCon (.dat + .xml) if necessary
        if (onSaveIconPerFrame != null) onSaveIconPerFrame(frameIndex, frame);

        // 2) Range component
        var range = frame.GetRange();
        if (range == null) return;

        // 3) Get IComponent (Raw U16 or Z[mm])
        var comp = useTransformed ? range.CopyTransformed()
                                : range.CopyUnpacked();

        uint w = comp.GetWidth();
        uint h = comp.GetDeliveredHeight();
        uint bpp = comp.GetBitsPerPixel();
        if (w == 0 || h == 0 || bpp == 0) return;

        // 4) Non-Image / Non-recommended API: Get raw data
        IntPtr ptr = comp.GetData();
        int bytes = checked((int)comp.GetDeliveredDataSize());
        if (ptr == IntPtr.Zero || bytes <= 0) return;

        var raw = new byte[bytes];
        Marshal.Copy(ptr, raw, 0, bytes);

        int ww = (int)w;
        int hh = (int)h;

        // 5) Stack into hm
        if (!useTransformed && bpp == 16)
        {
            // U16 -> float
            var u16 = new ushort[ww * hh];
            Buffer.BlockCopy(raw, 0, u16, 0, Math.Min(raw.Length, u16.Length * 2));

            for (int r = 0; r < hh && yWrite < lineCount; r++)
            {
                int off = r * ww;
                for (int x = 0; x < _width && x < ww; x++)
                    hm[yWrite, x] = u16[off + x]; // Apply scale/offset for mm conversion here if necessary
                yWrite++;
            }
        }
        else if (useTransformed && bpp == 64)
        {
            // double(mm) -> float
            var dbl = new double[ww * hh];
            Buffer.BlockCopy(raw, 0, dbl, 0, Math.Min(raw.Length, dbl.Length * 8));

            for (int r = 0; r < hh && yWrite < lineCount; r++)
            {
                int off = r * ww;
                for (int x = 0; x < _width && x < ww; x++)
                    hm[yWrite, x] = (float)dbl[off + x];
                yWrite++;
            }
        }
        else
        {
            // Add a branch here if other formats (e.g., 8/32bpp) arrive
        }
    }

    // ---- GenICam Node Helpers ----
    private void SetEnumMust(string name, string entry) { try { _p.SetEnum(name, entry); } catch (Exception ex) { throw new InvalidOperationException($"Failed to set {name}='{entry}': {ex.Message}"); } }
    private void SetFloatMust(string name, double v) { try { _p.SetFloat(name, (float)v); } catch (Exception ex) { throw new InvalidOperationException($"Failed to set {name}={v}: {ex.Message}"); } }
    private double GetFloatMust(string name) { try { return _p.GetFloat(name); } catch (Exception ex) { throw new InvalidOperationException($"Failed to read {name}: {ex.Message}"); } }
    private void TrySetEnum(string name, string entry) { try { _p.SetEnum(name, entry); } catch { } }
    private void TrySetFloat(string name, double v) { try { _p.SetFloat(name, (float)v); } catch { } }
    private void TryExec(string name) { try { _p.ExecuteCommand(name); } catch { } }

    // ---- Termination Process ----
    public void Close()
    {
        _grabber?.Dispose(); _grabber = null;
        _camera?.Dispose(); _camera = null;
        _discovery?.Dispose(); _discovery = null;
        _p = null;
    }
    public void Dispose() => Close();

    private void HeightCheckExample()
    {
        int lineCount = 2000; // User specified
        int linesPerFrame;

        // First, attempt the "setting value (HALCON equivalent)"
        if (!TryGetHeightNode(out linesPerFrame) || linesPerFrame <= 0)
        {
            // If unable to read, compensate with "actual measurement"
            TryProbeDeliveredHeight(out linesPerFrame, timeoutMsPerFrame);
        }

        // Final safeguard
        if (linesPerFrame <= 0) linesPerFrame = 1;

        // Required number of frames to satisfy the desired line count (Same as HALCON's formula)
        int loops = (lineCount + linesPerFrame - 1) / linesPerFrame;
    }

    // ① Read Height as a setting value from the GenICam node (HALCON equivalent)
    private bool TryGetHeightNode(out int height)
    {
        height = 0;
        try
        {
            double v = _p.GetFloat("Height"); // Replacement for GetFramegrabberParam("Height")
            height = (int)Math.Round(v);
            return height > 0;
        }
        catch { return false; }
    }

    // ② Measure Delivered line count from actual data (from the first arrived frame)
    private bool TryProbeDeliveredHeight(out int deliveredHeight, int timeoutMs)
    {
        deliveredHeight = 0;
        bool startedHere = false;
        if (!_grabber.IsStarted) { _grabber.Start(); startedHere = true; }

        try
        {
            var res = _grabber.GrabNext(TimeSpan.FromMilliseconds(timeoutMs));
            bool ok = false;
            int localHeight = 0; // ← Place outside the lambda

            res.IfCompleteFrame(frame =>
            {
                try
                {
                    var range = frame.GetRange();
                    if (range != null)
                    {
                        var comp = range.CopyUnpacked();
                        localHeight = (int)comp.GetDeliveredHeight();
                        ok = localHeight > 0;
                    }
                }
                finally { frame.Release(); }
            })
            .IfIncompleteFrame(f => { f.Release(); })
            .IfTimedOut(() => { });

            if (ok) deliveredHeight = localHeight; // ← Assign outside
            return ok;
        }
        finally
        {
            if (startedHere) _grabber.Stop();
        }
    }

        _ctl = new Ruler3000Lines();

        _ctl.ConfigureForExternalTrigger(exposureUs: 3000, triggerLine: "Line1");
    }

    void BtnGrab_Click()
    {

    }

}
