from pathlib import Path
from typing import Any

import numpy as np
import onnxruntime as ort
from PIL import Image, ImageDraw


class ImageInferenceProcessor:
    """
    ONNXモデルを1回だけ読み込み、
    PIL画像に対して繰り返し推論するクラス。
    """

    def __init__(
        self,
        model_path: str | Path,
    ) -> None:
        self._model_path = Path(model_path).resolve()

        if not self._model_path.exists():
            raise FileNotFoundError(
                f"ONNXモデルが見つかりません: {self._model_path}"
            )

        # __init__で1回だけモデルを読み込む
        self._session = ort.InferenceSession(
            str(self._model_path),
            providers=[
                "CPUExecutionProvider",
            ],
        )

        # モデルの入出力情報を保持
        self._input_info = self._session.get_inputs()[0]
        self._input_name = self._input_info.name

        self._output_infos = self._session.get_outputs()
        self._output_names = [
            output_info.name
            for output_info in self._output_infos
        ]

        # 実際のクラス定義に合わせて変更する
        self._class_names = [
            "OK",
            "NG",
        ]

    def process_image(
        self,
        image: Image.Image,
        camera_number: int,
        request_id: int | None = None,
    ) -> list[dict[str, Any]]:
        """
        PIL画像1枚を受け取り、加工画像と分類結果を返す。

        Parameters
        ----------
        image:
            C#から受信した画像を復元したPIL.Image.Image。

        camera_number:
            C#から受信したカメラ番号。

        request_id:
            C#側の要求を識別する任意の番号。

        Returns
        -------
        list[dict]:
            各要素は次の情報を持つ。

            {
                "image": PIL.Image.Image,
                "class_name": str,
                "score": float,
                "camera_number": int,
                "result_index": int,
                "request_id": int | None
            }
        """

        if not isinstance(image, Image.Image):
            raise TypeError(
                "imageはPIL.Image.Imageである必要があります。"
                f" actual={type(image)}"
            )

        if not isinstance(camera_number, int):
            raise TypeError(
                "camera_numberはintである必要があります。"
                f" actual={type(camera_number)}"
            )

        if camera_number < 0:
            raise ValueError(
                "camera_numberは0以上である必要があります。"
            )

        # Image.open()元のストリームから独立した画像を保持する
        source_image = image.copy()

        # カメラ番号によって設定を切り替える場合に使用
        camera_config = self._get_camera_config(
            camera_number
        )

        # 必要なら1枚の入力画像から複数の対象画像を作る
        target_images = self._create_target_images(
            source_image,
            camera_config,
        )

        results: list[dict[str, Any]] = []

        for result_index, target_image in enumerate(target_images):
            input_tensor = self._preprocess(
                target_image,
                camera_config,
            )

            class_name, score = self._run_inference(
                input_tensor
            )

            output_image = self._create_output_image(
                target_image,
                class_name=class_name,
                score=score,
                camera_number=camera_number,
            )

            results.append(
                {
                    # 通信直前まではPIL.Image.Imageで保持する
                    "image": output_image,

                    "class_name": str(class_name),
                    "score": float(score),

                    "camera_number": int(camera_number),
                    "result_index": int(result_index),
                    "request_id": request_id,
                }
            )

        return results

    def _get_camera_config(
        self,
        camera_number: int,
    ) -> dict[str, Any]:
        """
        カメラ番号ごとの設定を返す。

        実際の仕様に合わせて設定値を変更する。
        """

        camera_configs: dict[int, dict[str, Any]] = {
            0: {
                "input_width": 224,
                "input_height": 224,
                "rotation": 0,
            },
            1: {
                "input_width": 224,
                "input_height": 224,
                "rotation": 180,
            },
        }

        # 未登録カメラには標準設定を使用する例
        return camera_configs.get(
            camera_number,
            {
                "input_width": 224,
                "input_height": 224,
                "rotation": 0,
            },
        )

    def _create_target_images(
        self,
        image: Image.Image,
        camera_config: dict[str, Any],
    ) -> list[Image.Image]:
        """
        入力画像から、推論対象となる画像を複数生成する。

        HDevelopですでに切り出し済みなら、
        単純に1枚だけ返してもよい。
        """

        rotation = int(
            camera_config.get("rotation", 0)
        )

        if rotation != 0:
            image = image.rotate(
                rotation,
                expand=True,
            )

        # 現在は入力画像をそのまま1枚返す
        return [
            image.copy(),
        ]

    def _preprocess(
        self,
        image: Image.Image,
        camera_config: dict[str, Any],
    ) -> np.ndarray:
        """
        PIL画像をONNXモデルの入力テンソルへ変換する。

        この部分は学習時の前処理と完全に一致させる必要がある。
        """

        input_width = int(
            camera_config["input_width"]
        )
        input_height = int(
            camera_config["input_height"]
        )

        # モデルがRGB入力である例
        rgb_image = image.convert("RGB")

        resized_image = rgb_image.resize(
            (input_width, input_height),
            resample=Image.Resampling.BILINEAR,
        )

        # HWC、uint8、0～255
        image_array = np.asarray(
            resized_image,
            dtype=np.float32,
        )

        # 0～255から0.0～1.0へ
        image_array /= 255.0

        # HWCからCHWへ
        image_array = np.transpose(
            image_array,
            (2, 0, 1),
        )

        # CHWからNCHWへ
        image_array = np.expand_dims(
            image_array,
            axis=0,
        )

        return np.ascontiguousarray(
            image_array,
            dtype=np.float32,
        )

    def _run_inference(
        self,
        input_tensor: np.ndarray,
    ) -> tuple[str, float]:
        """
        ONNX推論を実行して、クラス名とスコアを返す。

        出力形式は実際のONNXモデルに合わせて変更する。
        """

        outputs = self._session.run(
            self._output_names,
            {
                self._input_name: input_tensor,
            },
        )

        # 分類モデルの出力が shape=(1, class_count) の
        # logitsまたは確率である例
        scores = np.asarray(
            outputs[0]
        ).squeeze()

        if scores.ndim != 1:
            raise ValueError(
                "想定外のONNX出力形状です。"
                f" shape={scores.shape}"
            )

        class_index = int(
            np.argmax(scores)
        )

        score = float(
            scores[class_index]
        )

        if not 0 <= class_index < len(self._class_names):
            raise IndexError(
                "ONNX出力のクラス番号が範囲外です。"
                f" class_index={class_index}"
            )

        class_name = self._class_names[class_index]

        return class_name, score

    def _create_output_image(
        self,
        image: Image.Image,
        class_name: str,
        score: float,
        camera_number: int,
    ) -> Image.Image:
        """
        C#へ返す加工済みPIL画像を生成する。

        元画像そのものを返す場合でもcopy()しておく。
        """

        output_image = image.convert("RGB").copy()

        draw = ImageDraw.Draw(output_image)

        label = (
            f"Camera={camera_number} "
            f"Class={class_name} "
            f"Score={score:.4f}"
        )

        # 簡易的なラベル背景
        draw.rectangle(
            (
                0,
                0,
                min(output_image.width, 400),
                32,
            ),
            fill=(0, 0, 0),
        )

        draw.text(
            (5, 7),
            label,
            fill=(255, 255, 255),
        )

        return output_image
