    using MicrexSxLoaderCommand.Core.Communication;
using MicrexSxLoaderCommand.Core.Memory;
using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Windows.Forms;

namespace MicrexSxLoaderCommand
{
    public partial class MainForm : Form
    {
        /// <summary>
        /// Name of the XML address map file located next to the executable.
        /// </summary>
        private readonly string XmlFileName = "AddressMap.xml";

        /// <summary>
        /// TCP server that handles loader commands.
        /// </summary>
        private LoaderCommandTcpServer _server;

        /// <summary>
        /// In-memory representation of PLC WM words.
        /// </summary>
        private PlcMemory _memory = new PlcMemory();

        /// <summary>
        /// Address map model loaded from XML.
        /// </summary>
        private MicrexSxLoaderCommand.Core.AddressMap.AddressMapModel _map;

        /// <summary>
        /// BindingSource for bit rows grid.
        /// </summary>
        private BindingSource _bitsSource = new BindingSource();

        /// <summary>
        /// BindingSource for word rows grid.
        /// </summary>
        private BindingSource _wordsSource = new BindingSource();

        /// <summary>
        /// Rows representing bit items in the grid.
        /// </summary>
        private List<BitRow> _bitRows = new List<BitRow>();

        /// <summary>
        /// Rows representing word items in the grid.
        /// </summary>
        private List<WordRow> _wordRows = new List<WordRow>();

        /// <summary>
        /// Persistence helper for saving/loading simulated state.
        /// </summary>
        private MicrexSxLoaderCommand.Core.Persistence.SimStateStore _stateStore
    = new MicrexSxLoaderCommand.Core.Persistence.SimStateStore();

        /// <summary>
        /// Path of the saved state file.
        /// </summary>
        private string _statePath;



        /// <summary>
        /// Default constructor. Initializes components, server and event handlers.
        /// </summary>
        public MainForm()
        {
            InitializeComponent();

            btnStop.Enabled = false;

            _server = new LoaderCommandTcpServer(_memory);
            _server.Log += OnServerLog;

            // When core memory is updated by network activity, refresh UI grid values safely on the UI thread.
            _server.MemoryUpdated += () =>
            {
                if (IsDisposed) return;
                BeginInvoke(new Action(RefreshGridValues));
            };

        }

        /// <summary>
        /// Refresh displayed Value cells in the grids (bits and words) from PlcMemory.
        /// </summary>
        private void RefreshGridValues()
        {
            // Update bit rows from memory
            for (int i = 0; i < _bitRows.Count; i++)
            {
                var r = _bitRows[i];
                r.Value = ReadBitValue(r.Addr, r.Bit);
            }

            // Update word rows from memory
            for (int i = 0; i < _wordRows.Count; i++)
            {
                var r = _wordRows[i];
                if (r.Type == "DWORD")
                    r.Value = _memory.ReadInt32_LowHigh(r.Addr);   // ★ReadInt32
                else
                    r.Value = _memory.ReadWm(r.Addr);
            }


            _bitsSource.ResetBindings(false);
            _wordsSource.ResetBindings(false);
        }



        /// <summary>
        /// Build grid rows based on the loaded address map model.
        /// </summary>
        private void BuildRowsFromMap()
        {
            _bitRows.Clear();
            _wordRows.Clear();

            // Bit rows
            foreach (var b in _map.Bits)
            {
                _bitRows.Add(new BitRow
                {
                    Group = b.GroupName,
                    Addr = b.WordAddress,
                    Bit = b.BitIndex,
                    Id = b.Id,
                    Description = b.Description,
                    Value = ReadBitValue(b.WordAddress, b.BitIndex)
                });
            }

            // Word rows (WORD + DWORD を表示)
            foreach (var w in _map.Words)
            {
                var t = (w.Type ?? "").Trim().ToUpperInvariant();
                if (t != "WORD" && t != "DWORD")
                    continue;

                int v;
                if (t == "WORD")
                    v = _memory.ReadWm(w.WordAddress);              // ushort → int
                else
                    v = _memory.ReadInt32_LowHigh(w.WordAddress);   // ★ ReadInt32

                _wordRows.Add(new WordRow
                {
                    Addr = w.WordAddress,
                    Id = w.Id,
                    Description = w.Description,
                    Type = t,
                    Value = v
                });
            }


            _bitsSource.DataSource = _bitRows;
            _wordsSource.DataSource = _wordRows;

            dgvBits.AutoGenerateColumns = true;
            dgvWords.AutoGenerateColumns = true;
            dgvBits.DataSource = _bitsSource;
            dgvWords.DataSource = _wordsSource;

            // Adjust column sizing behavior
            //dgvBits.AutoSizeColumnsMode = DataGridViewAutoSizeColumnsMode.AllCells;
            //dgvWords.AutoSizeColumnsMode = DataGridViewAutoSizeColumnsMode.AllCells;

            dgvBits.AutoSizeColumnsMode = DataGridViewAutoSizeColumnsMode.Fill;
            dgvWords.AutoSizeColumnsMode = DataGridViewAutoSizeColumnsMode.Fill;

            // Hide the row header (left-most margin)
            dgvBits.RowHeadersVisible = false;
            dgvWords.RowHeadersVisible = false;

            // Configure column editability so that only the Value column is editable
            SetGridEditability();
        }

        /// <summary>
        /// Configure per-column editability for DataGridViews: only the "Value" column is editable.
        /// </summary>
        private void SetGridEditability()
        {
            // Bits grid: make DataGridView editable at control level, then lock all columns
            dgvBits.ReadOnly = false; // Keep DataGridView editable so specific columns can be unlocked
            foreach (DataGridViewColumn c in dgvBits.Columns)
            {
                c.ReadOnly = true; // Lock all columns first
            }

            // Allow editing only for the Value column (checkbox)
            var bitsValueCol = dgvBits.Columns.Cast<DataGridViewColumn>()
                .FirstOrDefault(c => c.DataPropertyName == "Value");
            if (bitsValueCol != null)
                bitsValueCol.ReadOnly = false;

            // Words grid
            dgvWords.ReadOnly = false;
            foreach (DataGridViewColumn c in dgvWords.Columns)
            {
                c.ReadOnly = true; // Lock all columns
            }

            // Allow editing only for the Value column (numeric)
            var wordsValueCol = dgvWords.Columns.Cast<DataGridViewColumn>()
                .FirstOrDefault(c => c.DataPropertyName == "Value");
            if (wordsValueCol != null)
                wordsValueCol.ReadOnly = false;

            // Prevent user from adding rows
            dgvBits.AllowUserToAddRows = false;
            dgvWords.AllowUserToAddRows = false;

            // Prevent user from deleting rows
            dgvBits.AllowUserToDeleteRows = false;
            dgvWords.AllowUserToDeleteRows = false;

            dgvBits.EditMode = DataGridViewEditMode.EditOnEnter;
            dgvWords.EditMode = DataGridViewEditMode.EditOnEnter;

        }



        /// <summary>
        /// Read a single bit from a WM word in memory.
        /// </summary>
        /// <param name="wordAddr">Word address (index).</param>
        /// <param name="bitIndex">Bit index (0..15).</param>
        /// <returns>True when the bit is set; otherwise false.</returns>
        private bool ReadBitValue(int wordAddr, int bitIndex)
        {
            ushort word = _memory.ReadWm(wordAddr);
            int mask = 1 << bitIndex;
            return (word & mask) != 0;
        }



        private void btnStart_Click(object sender, EventArgs e)
        {
            try
            {
                // Start server on port 507 by default (loopback/local testing is simplest)
                _server.Start(port: 507);

                btnStart.Enabled = false;
                btnStop.Enabled = true;

                AppendLog("[UI] Server started.");
            }
            catch (Exception ex)
            {
                AppendLog("[UI] Start error: " + ex.Message);
            }
        }

        private void btnStop_Click(object sender, EventArgs e)
        {
            try
            {
                _server.Stop();

                btnStart.Enabled = true;
                btnStop.Enabled = false;

                AppendLog("[UI] Server stopped.");
            }
            catch (Exception ex)
            {
                AppendLog("[UI] Stop error: " + ex.Message);
            }
        }

        /// <summary>
        /// Save simulated state and dispose server when the form is closing.
        /// </summary>
        private void MainForm_FormClosing(object sender, FormClosingEventArgs e)
        {
            try
            {
                if (string.IsNullOrEmpty(_statePath))
                    _statePath = _stateStore.GetDefaultPath();

                var snap = _memory.GetWmSnapshot();
                _stateStore.Save(_statePath, snap);

                AppendLog("[STATE] Saved: " + _statePath);

                _server?.Dispose();
            }
            catch { /* ignore */ }
        }

        /// <summary>
        /// Handler for server log events. Routes messages to the UI logger.
        /// </summary>
        private void OnServerLog(string message)
        {
            // Server events may arrive on a non-UI thread, so append via UI-safe helper
            AppendLog(message);
        }

        /// <summary>
        /// Append a line to the UI log textbox in a thread-safe manner.
        /// </summary>
        /// <param name="text">Text to append.</param>
        private void AppendLog(string text)
        {
            if (this.IsDisposed) return;

            if (txtLog.InvokeRequired)
            {
                txtLog.BeginInvoke(new Action(() => AppendLog(text)));
                return;
            }

            txtLog.AppendText(text + Environment.NewLine);
        }


        private void dgvWords_CellEndEdit(object sender, DataGridViewCellEventArgs e)
        {
            if (e.RowIndex < 0) return;

            var col = dgvWords.Columns[e.ColumnIndex];
            if (col == null || col.DataPropertyName != "Value") return;

            var row = _wordRows[e.RowIndex];

            // Re-read cell value from grid (it may be boxed as object)
            object cellValue = dgvWords.Rows[e.RowIndex].Cells[e.ColumnIndex].Value;

            int newValue;
            if (!TryParseInt32(cellValue, out newValue))
            {
                MessageBox.Show("Please enter a valid integer value.");
                dgvWords.Rows[e.RowIndex].Cells[e.ColumnIndex].Value = row.Value;
                return;
            }

            if (row.Type == "WORD")
            {
                if (newValue < 0 || newValue > 65535)
                {
                    MessageBox.Show("WORD must be between 0 and 65535.");
                    dgvWords.Rows[e.RowIndex].Cells[e.ColumnIndex].Value = row.Value;
                    return;
                }

                row.Value = newValue;
                _memory.WriteWm(row.Addr, (ushort)newValue);                 // WORD書き込み
                AppendLog($"[UI] WORD set addr={row.Addr} => {newValue} (0x{newValue:X4})");
            }
            else // DWORD
            {
                row.Value = newValue;
                _memory.WriteInt32_LowHigh(row.Addr, newValue);              // ★ここで WriteInt32
                AppendLog($"[UI] DWORD set addr={row.Addr} => {newValue}");
            }

            RefreshGridValues();

        }

        //new method
        private bool TryParseInt32(object value, out int result)
        {
            result = 0;
            if (value == null) return false;

            if (value is int)
            {
                result = (int)value;
                return true;
            }

            var s = value.ToString();
            if (string.IsNullOrWhiteSpace(s)) return false;

            return int.TryParse(s.Trim(), out result);
        }


        /// <summary>
        /// Try to parse an object value into ushort. Accepts ushort directly or decimal string.
        /// </summary>
        private bool TryParseUShort(object value, out ushort result)
        {
            result = 0;
            if (value == null) return false;

            // If it's already ushort
            if (value is ushort)
            {
                result = (ushort)value;
                return true;
            }

            // Try string representation
            var s = value.ToString();
            if (string.IsNullOrWhiteSpace(s)) return false;

            // Allow decimal only (0xFFFF hex could be added later)
            int n;
            if (!int.TryParse(s.Trim(), out n)) return false;
            if (n < 0 || n > 65535) return false;

            result = (ushort)n;
            return true;
        }


        /// <summary>
        /// Represents a row in the bit grid.
        /// </summary>
        public sealed class BitRow
        {
            public string Group { get; set; }
            public int Addr { get; set; }
            public int Bit { get; set; }
            public string Id { get; set; }
            public string Description { get; set; }

            /// <summary>
            /// Current boolean value for UI display.
            /// </summary>
            public bool Value { get; set; }
        }

        /// <summary>
        /// Represents a row in the word grid.
        /// </summary>
        public sealed class WordRow
        {
            public int Addr { get; set; }
            public string Id { get; set; }
            public string Description { get; set; }

            public string Type { get; set; }            //Add
            /// <summary>
            /// Current ushort value for UI display.
            /// </summary>
            public int Value { get; set; }              //ushort 2 int
        }

        /// <summary>
        /// Form load handler: load XML map, restore saved state, build grids and attach UI events.
        /// </summary>
        private void MainForm_Load(object sender, EventArgs e)
        {
            try
            {
                // 1) Load XML address map definition
                LoadXML();

                // 2) Determine default path for state file
                _statePath = _stateStore.GetDefaultPath();

                // 3) Load JSON saved state and restore into PlcMemory
                try
                {
                    var snap = _stateStore.Load(_statePath);
                    _memory.LoadWmSnapshot(snap);
                    AppendLog("[STATE] Loaded: " + _statePath);
                }
                catch (Exception ex)
                {
                    AppendLog("[STATE] Load ERROR: " + ex.Message);
                }

                // 4) Build grid rows from address map (memory already restored so values reflect saved state)
                BuildRowsFromMap();
                RefreshGridValues();

                // 5) Attach grid event handlers after initial render to avoid spurious events during startup
                dgvBits.CurrentCellDirtyStateChanged += dgvBits_CurrentCellDirtyStateChanged;
                dgvBits.CellValueChanged += dgvBits_CellValueChanged;
                dgvWords.CellEndEdit += dgvWords_CellEndEdit;
            }
            catch (Exception ex)
            {
                AppendLog("Load ERROR: " + ex.Message);
            }
        }


        /// <summary>
        /// Load address map XML from the executable directory.
        /// </summary>
        private void LoadXML()
        {
            try
            {
                string baseDir = AppDomain.CurrentDomain.BaseDirectory; // location of the executable
                string xmlPath = Path.Combine(baseDir, XmlFileName);

                var loader = new MicrexSxLoaderCommand.Core.AddressMap.AddressMapXmlLoader();
                _map = loader.LoadFromFile(xmlPath);

                AppendLog(string.Format("AddressMap loaded. Bits={0}, Words={1}", _map.Bits.Count, _map.Words.Count));

            }
            catch (Exception ex)
            {
                AppendLog("XML Load ERROR: " + ex.Message);
            }
        }


        private void dgvBits_CurrentCellDirtyStateChanged(object sender, EventArgs e)
        {
            // Commit checkbox changes immediately so CellValueChanged fires right away
            if (dgvBits.IsCurrentCellDirty)
            {
                dgvBits.CommitEdit(DataGridViewDataErrorContexts.Commit);
            }
        }

        private void dgvBits_CellValueChanged(object sender, DataGridViewCellEventArgs e)
        {
            if (e.RowIndex < 0) return;

            // Only handle changes to the Value column (auto-generated column name is "Value")
            var col = dgvBits.Columns[e.ColumnIndex];
            if (col == null || col.DataPropertyName != "Value") return;

            var row = _bitRows[e.RowIndex];
            WriteBitValue(row.Addr, row.Bit, row.Value);

            AppendLog(string.Format("[UI] BIT set addr={0} bit={1} => {2}", row.Addr, row.Bit, row.Value));
            RefreshGridValues();
        }

        /// <summary>
        /// Write a single bit in the specified WM word address.
        /// </summary>
        /// <param name="wordAddr">Word address (index).</param>
        /// <param name="bitIndex">Bit index (0..15).</param>
        /// <param name="on">True to set the bit; false to clear it.</param>
        private void WriteBitValue(int wordAddr, int bitIndex, bool on)
        {
            ushort word = _memory.ReadWm(wordAddr);
            int mask = 1 << bitIndex;

            ushort newWord;
            if (on)
                newWord = (ushort)(word | mask);
            else
                newWord = (ushort)(word & ~mask);

            _memory.WriteWm(wordAddr, newWord);
        }
    }
