    namespace ImageRecognition.PythonCommunication
    {
        /// <summary>
        /// Pythonから返されるJSON全体。
        /// </summary>
        public sealed class PythonResponseMetadata
        {
            public PythonResponseMetadata()
            {
                Results = new List<PythonResultMetadata>();
            }

            [JsonProperty("success")]
            public bool Success { get; set; }

            [JsonProperty("request_id")]
            public long? RequestId { get; set; }

            [JsonProperty("result_count")]
            public int ResultCount { get; set; }

            [JsonProperty("results")]
            public List<PythonResultMetadata> Results { get; set; }

            [JsonProperty("shutdown")]
            public bool Shutdown { get; set; }

            [JsonProperty("error")]
            public PythonErrorMetadata Error { get; set; }
        }

        /// <summary>
        /// Pythonから返される1画像分のメタデータ。
        /// </summary>
        public sealed class PythonResultMetadata
        {
            public PythonResultMetadata()
            {
                ClassName = string.Empty;
                ImageFormat = string.Empty;
                ImageMode = string.Empty;
            }

            [JsonProperty("result_index")]
            public int ResultIndex { get; set; }

            [JsonProperty("class_name")]
            public string ClassName { get; set; }

            [JsonProperty("score")]
            public double Score { get; set; }

            [JsonProperty("camera_number")]
            public int? CameraNumber { get; set; }

            [JsonProperty("image_index")]
            public int ImageIndex { get; set; }

            [JsonProperty("image_format")]
            public string ImageFormat { get; set; }

            [JsonProperty("image_length")]
            public long ImageLength { get; set; }

            [JsonProperty("image_width")]
            public int ImageWidth { get; set; }

            [JsonProperty("image_height")]
            public int ImageHeight { get; set; }

            [JsonProperty("image_mode")]
            public string ImageMode { get; set; }
        }

        public sealed class PythonErrorMetadata
        {
            public PythonErrorMetadata()
            {
                Code = string.Empty;
                Message = string.Empty;
            }

            [JsonProperty("code")]
            public string Code { get; set; }

            [JsonProperty("message")]
            public string Message { get; set; }
        }

        /// <summary>
        /// C#側で最終的に使用する1結果分のデータ。
        /// </summary>
        public sealed class PythonInferenceResult
        {
            public PythonInferenceResult()
            {
                ClassName = string.Empty;
                ImageFormat = string.Empty;
                ImageMode = string.Empty;
                PngBytes = new byte[0];
            }

            public int ResultIndex { get; set; }

            public string ClassName { get; set; }

            public double Score { get; set; }

            public int? CameraNumber { get; set; }

            public string ImageFormat { get; set; }

            public int ImageWidth { get; set; }

            public int ImageHeight { get; set; }

            public string ImageMode { get; set; }

            /// <summary>
            /// PNGファイル形式のバイト列。
            /// 生画素データではない。
            /// </summary>
            public byte[] PngBytes { get; set; }
        }

        /// <summary>
        /// C#側で使用する推論応答全体。
        /// </summary>
        public sealed class PythonInferenceResponse
        {
            public PythonInferenceResponse()
            {
                Results = new List<PythonInferenceResult>();
            }

            public long? RequestId { get; set; }

            public IList<PythonInferenceResult> Results { get; set; }
        }
    }

    using System;

    namespace ImageRecognition.PythonCommunication
    {
        public sealed class PythonInferenceException : Exception
        {
            public PythonInferenceException(
                string errorCode,
                string message,
                long? requestId)
                : base(message)
            {
                ErrorCode = errorCode;
                RequestId = requestId;
            }

            public string ErrorCode { get; private set; }

            public long? RequestId { get; private set; }

            public override string ToString()
            {
                return
                    base.ToString()
                    + Environment.NewLine
                    + "PythonErrorCode: "
                    + ErrorCode
                    + Environment.NewLine
                    + "RequestId: "
                    + (RequestId.HasValue
                        ? RequestId.Value.ToString()
                        : "(null)");
            }
        }
    }



    //Python常駐プロセスとの通信クラス

    using System;
    using System.Collections.Generic;
    using System.Diagnostics;
    using System.IO;
    using System.Linq;
    using System.Text;
    using System.Threading;
    using System.Threading.Tasks;
    using Newtonsoft.Json;

    namespace ImageRecognition.PythonCommunication
    {
        public sealed class PythonInferenceClient : IDisposable
        {
            private const long MaximumJsonBytes =
                10L * 1024L * 1024L;

            private const long MaximumImageBytes =
                512L * 1024L * 1024L;

            private const int MaximumResultImageCount =
                10000;

            private readonly string _pythonExecutablePath;
            private readonly string _communicationMainPath;
            private readonly string _workingDirectory;

            /*
            * 現在の通信仕様は、
            * 1要求を送ったらその応答を最後まで受け取ってから
            * 次の要求を送る逐次通信。
            *
            * 複数スレッドから同時に呼ばれても
            * 通信データが混ざらないように直列化する。
            */
            private readonly SemaphoreSlim _communicationLock;

            private Process _process;
            private Stream _pythonInput;
            private Stream _pythonOutput;

            private Task _standardErrorReadTask;

            private long _nextRequestId;
            private bool _disposed;

            public PythonInferenceClient(
                string pythonExecutablePath,
                string communicationMainPath,
                string workingDirectory)
            {
                if (pythonExecutablePath == null)
                {
                    throw new ArgumentNullException(
                        "pythonExecutablePath");
                }

                if (communicationMainPath == null)
                {
                    throw new ArgumentNullException(
                        "communicationMainPath");
                }

                if (workingDirectory == null)
                {
                    throw new ArgumentNullException(
                        "workingDirectory");
                }

                _pythonExecutablePath =
                    pythonExecutablePath;

                _communicationMainPath =
                    communicationMainPath;

                _workingDirectory =
                    workingDirectory;

                _communicationLock =
                    new SemaphoreSlim(1, 1);
            }

            /// <summary>
            /// Pythonのstderrログ通知。
            /// </summary>
            public event EventHandler<string> PythonLogReceived;

            /// <summary>
            /// Pythonプロセスの予期しない終了通知。
            /// </summary>
            public event EventHandler PythonProcessExited;

            public bool IsRunning
            {
                get
                {
                    return
                        _process != null
                        && !_process.HasExited;
                }
            }

            /// <summary>
            /// communication_main.pyを1回だけ起動する。
            /// </summary>
            public Task StartAsync()
            {
                ThrowIfDisposed();

                if (IsRunning)
                {
                    return Task.FromResult(0);
                }

                ValidatePaths();

                ProcessStartInfo startInfo =
                    new ProcessStartInfo();

                startInfo.FileName =
                    _pythonExecutablePath;

                /*
                * .NET Framework 4.8ではArgumentListがないため、
                * スクリプトパスをダブルクォートで囲む。
                */
                startInfo.Arguments =
                    "\"" + _communicationMainPath + "\"";

                startInfo.WorkingDirectory =
                    _workingDirectory;

                startInfo.UseShellExecute = false;
                startInfo.CreateNoWindow = true;

                startInfo.RedirectStandardInput = true;
                startInfo.RedirectStandardOutput = true;
                startInfo.RedirectStandardError = true;

                _process = new Process();
                _process.StartInfo = startInfo;
                _process.EnableRaisingEvents = true;

                _process.Exited +=
                    Process_Exited;

                bool started = _process.Start();

                if (!started)
                {
                    _process.Dispose();
                    _process = null;

                    throw new InvalidOperationException(
                        "Pythonプロセスを起動できませんでした。");
                }

                /*
                * JSONだけでなくPNGバイナリも扱うため、
                * StandardInput/OutputのBaseStreamを使用する。
                */
                _pythonInput =
                    _process.StandardInput.BaseStream;

                _pythonOutput =
                    _process.StandardOutput.BaseStream;

                /*
                * stderrを常時読み続ける。
                *
                * stderrを読まないと、バッファが満杯になった際に
                * Python側が停止する可能性がある。
                */
                _standardErrorReadTask =
                    ReadStandardErrorLoopAsync(
                        _process);

                return Task.FromResult(0);
            }

            /// <summary>
            /// PNG画像1枚とカメラ番号をPythonへ送り、
            /// 分類結果と複数PNGを受け取る。
            /// </summary>
            public async Task<PythonInferenceResponse>
                ProcessImageAsync(
                    byte[] inputPngBytes,
                    int cameraNumber,
                    CancellationToken cancellationToken)
            {
                if (inputPngBytes == null)
                {
                    throw new ArgumentNullException(
                        "inputPngBytes");
                }

                if (inputPngBytes.Length == 0)
                {
                    throw new ArgumentException(
                        "入力PNGバイト列が空です。",
                        "inputPngBytes");
                }

                if (cameraNumber < 0)
                {
                    throw new ArgumentOutOfRangeException(
                        "cameraNumber",
                        "カメラ番号は0以上である必要があります。");
                }

                ThrowIfDisposed();
                EnsureRunning();

                long requestId =
                    Interlocked.Increment(
                        ref _nextRequestId);

                PythonProcessRequest request =
                    new PythonProcessRequest();

                request.Command = "process";
                request.RequestId = requestId;
                request.CameraNumber = cameraNumber;

                string requestJson =
                    JsonConvert.SerializeObject(
                        request,
                        Formatting.None);

                byte[] requestJsonBytes =
                    Encoding.UTF8.GetBytes(
                        requestJson);

                await _communicationLock.WaitAsync(
                    cancellationToken);

                try
                {
                    EnsureRunning();

                    await WriteRequestAsync(
                        requestJsonBytes,
                        inputPngBytes,
                        cancellationToken);

                    PythonInferenceResponse response =
                        await ReadInferenceResponseAsync(
                            requestId,
                            cancellationToken);

                    return response;
                }
                catch (IOException exception)
                {
                    throw new InvalidOperationException(
                        "Pythonとの標準入出力通信に失敗しました。",
                        exception);
                }
                finally
                {
                    _communicationLock.Release();
                }
            }

            /// <summary>
            /// CancellationTokenを省略して呼ぶためのオーバーロード。
            /// </summary>
            public Task<PythonInferenceResponse>
                ProcessImageAsync(
                    byte[] inputPngBytes,
                    int cameraNumber)
            {
                return ProcessImageAsync(
                    inputPngBytes,
                    cameraNumber,
                    CancellationToken.None);
            }

            private async Task WriteRequestAsync(
                byte[] jsonBytes,
                byte[] imageBytes,
                CancellationToken cancellationToken)
            {
                if (_pythonInput == null)
                {
                    throw new InvalidOperationException(
                        "Python標準入力が初期化されていません。");
                }

                /*
                * C# → Python
                *
                * 8 bytes: JSON長
                * N bytes: JSON
                * 8 bytes: PNG長
                * M bytes: PNG
                */
                await WriteUInt64Async(
                    _pythonInput,
                    checked((ulong)jsonBytes.LongLength),
                    cancellationToken);

                await _pythonInput.WriteAsync(
                    jsonBytes,
                    0,
                    jsonBytes.Length,
                    cancellationToken);

                await WriteUInt64Async(
                    _pythonInput,
                    checked((ulong)imageBytes.LongLength),
                    cancellationToken);

                if (imageBytes.Length > 0)
                {
                    await _pythonInput.WriteAsync(
                        imageBytes,
                        0,
                        imageBytes.Length,
                        cancellationToken);
                }

                await _pythonInput.FlushAsync(
                    cancellationToken);
            }

            private async Task<PythonInferenceResponse>
                ReadInferenceResponseAsync(
                    long expectedRequestId,
                    CancellationToken cancellationToken)
            {
                if (_pythonOutput == null)
                {
                    throw new InvalidOperationException(
                        "Python標準出力が初期化されていません。");
                }

                /*
                * Python → C#
                *
                * 8 bytes: 結果JSON長
                * N bytes: 結果JSON
                * 8 bytes: 画像枚数
                *
                * 各画像:
                *   8 bytes: PNG長
                *   M bytes: PNG
                */
                ulong jsonLengthValue =
                    await ReadUInt64Async(
                        _pythonOutput,
                        cancellationToken);

                int jsonLength =
                    ValidateAndConvertLength(
                        jsonLengthValue,
                        MaximumJsonBytes,
                        "応答JSON");

                byte[] jsonBytes =
                    new byte[jsonLength];

                await ReadExactlyAsync(
                    _pythonOutput,
                    jsonBytes,
                    0,
                    jsonBytes.Length,
                    cancellationToken);

                string responseJson =
                    Encoding.UTF8.GetString(
                        jsonBytes);

                PythonResponseMetadata metadata;

                try
                {
                    metadata =
                        JsonConvert.DeserializeObject
                        <PythonResponseMetadata>(
                            responseJson);
                }
                catch (JsonException exception)
                {
                    throw new InvalidDataException(
                        "Python応答JSONを解析できません。"
                        + Environment.NewLine
                        + responseJson,
                        exception);
                }

                if (metadata == null)
                {
                    throw new InvalidDataException(
                        "Python応答JSONがnullです。");
                }

                ulong imageCountValue =
                    await ReadUInt64Async(
                        _pythonOutput,
                        cancellationToken);

                if (imageCountValue >
                    (ulong)MaximumResultImageCount)
                {
                    throw new InvalidDataException(
                        "結果画像数が上限を超えています。"
                        + " Count="
                        + imageCountValue);
                }

                int imageCount =
                    checked((int)imageCountValue);

                List<byte[]> imageBytesList =
                    new List<byte[]>(
                        imageCount);

                for (int index = 0;
                    index < imageCount;
                    index++)
                {
                    ulong imageLengthValue =
                        await ReadUInt64Async(
                            _pythonOutput,
                            cancellationToken);

                    int imageLength =
                        ValidateAndConvertLength(
                            imageLengthValue,
                            MaximumImageBytes,
                            "結果画像[" + index + "]");

                    byte[] imageBytes =
                        new byte[imageLength];

                    await ReadExactlyAsync(
                        _pythonOutput,
                        imageBytes,
                        0,
                        imageBytes.Length,
                        cancellationToken);

                    imageBytesList.Add(
                        imageBytes);
                }

                /*
                * 応答をすべて読み取ってから、
                * Python側エラーを例外化する。
                *
                * 途中で読み取りを止めると
                * 次の応答境界が崩れるため。
                */
                if (!metadata.Success)
                {
                    string errorCode =
                        metadata.Error != null
                        ? metadata.Error.Code
                        : "UNKNOWN_ERROR";

                    string errorMessage =
                        metadata.Error != null
                        ? metadata.Error.Message
                        : "Python側で不明なエラーが発生しました。";

                    throw new PythonInferenceException(
                        errorCode,
                        errorMessage,
                        metadata.RequestId);
                }

                if (!metadata.RequestId.HasValue)
                {
                    throw new InvalidDataException(
                        "Python応答にrequest_idがありません。");
                }

                if (metadata.RequestId.Value !=
                    expectedRequestId)
                {
                    throw new InvalidDataException(
                        "request_idが一致しません。"
                        + " Expected="
                        + expectedRequestId
                        + ", Actual="
                        + metadata.RequestId.Value);
                }

                if (metadata.Results == null)
                {
                    throw new InvalidDataException(
                        "Python応答のresultsがnullです。");
                }

                if (metadata.ResultCount !=
                    metadata.Results.Count)
                {
                    throw new InvalidDataException(
                        "result_countとresults件数が一致しません。"
                        + " ResultCount="
                        + metadata.ResultCount
                        + ", Results.Count="
                        + metadata.Results.Count);
                }

                if (metadata.ResultCount !=
                    imageCount)
                {
                    throw new InvalidDataException(
                        "result_countと結果画像数が一致しません。"
                        + " ResultCount="
                        + metadata.ResultCount
                        + ", ImageCount="
                        + imageCount);
                }

                List<PythonInferenceResult> results =
                    new List<PythonInferenceResult>(
                        metadata.Results.Count);

                foreach (
                    PythonResultMetadata resultMetadata
                    in metadata.Results.OrderBy(
                        item => item.ResultIndex))
                {
                    if (resultMetadata.ImageIndex < 0 ||
                        resultMetadata.ImageIndex >=
                        imageBytesList.Count)
                    {
                        throw new InvalidDataException(
                            "image_indexが範囲外です。"
                            + " ImageIndex="
                            + resultMetadata.ImageIndex
                            + ", ImageCount="
                            + imageBytesList.Count);
                    }

                    byte[] pngBytes =
                        imageBytesList[
                            resultMetadata.ImageIndex];

                    if (resultMetadata.ImageLength !=
                        pngBytes.LongLength)
                    {
                        throw new InvalidDataException(
                            "JSON上の画像長と実データ長が"
                            + "一致しません。"
                            + " ResultIndex="
                            + resultMetadata.ResultIndex
                            + ", Expected="
                            + resultMetadata.ImageLength
                            + ", Actual="
                            + pngBytes.LongLength);
                    }

                    if (!string.Equals(
                            resultMetadata.ImageFormat,
                            "png",
                            StringComparison.OrdinalIgnoreCase))
                    {
                        throw new InvalidDataException(
                            "未対応の画像形式です。"
                            + " Format="
                            + resultMetadata.ImageFormat);
                    }

                    PythonInferenceResult result =
                        new PythonInferenceResult();

                    result.ResultIndex =
                        resultMetadata.ResultIndex;

                    result.ClassName =
                        resultMetadata.ClassName;

                    result.Score =
                        resultMetadata.Score;

                    result.CameraNumber =
                        resultMetadata.CameraNumber;

                    result.ImageFormat =
                        resultMetadata.ImageFormat;

                    result.ImageWidth =
                        resultMetadata.ImageWidth;

                    result.ImageHeight =
                        resultMetadata.ImageHeight;

                    result.ImageMode =
                        resultMetadata.ImageMode;

                    result.PngBytes =
                        pngBytes;

                    results.Add(
                        result);
                }

                PythonInferenceResponse response =
                    new PythonInferenceResponse();

                response.RequestId =
                    metadata.RequestId;

                response.Results =
                    results;

                return response;
            }

            /// <summary>
            /// Pythonへshutdown要求を送る。
            /// </summary>
            public async Task ShutdownAsync()
            {
                if (_disposed)
                {
                    return;
                }

                if (!IsRunning)
                {
                    return;
                }

                await _communicationLock.WaitAsync();

                try
                {
                    if (!IsRunning)
                    {
                        return;
                    }

                    long requestId =
                        Interlocked.Increment(
                            ref _nextRequestId);

                    PythonShutdownRequest request =
                        new PythonShutdownRequest();

                    request.Command = "shutdown";
                    request.RequestId = requestId;

                    string json =
                        JsonConvert.SerializeObject(
                            request,
                            Formatting.None);

                    byte[] jsonBytes =
                        Encoding.UTF8.GetBytes(
                            json);

                    /*
                    * shutdownでは画像長0を送る。
                    */
                    await WriteRequestAsync(
                        jsonBytes,
                        new byte[0],
                        CancellationToken.None);

                    await ReadShutdownResponseAsync(
                        requestId,
                        CancellationToken.None);
                }
                catch
                {
                    /*
                    * 正常終了要求に失敗した場合は、
                    * Dispose()で強制終了する。
                    */
                }
                finally
                {
                    _communicationLock.Release();
                }

                if (_process != null &&
                    !_process.HasExited)
                {
                    bool exited =
                        await WaitForExitWithTimeoutAsync(
                            _process,
                            TimeSpan.FromSeconds(5));

                    if (!exited &&
                        !_process.HasExited)
                    {
                        _process.Kill();
                        _process.WaitForExit();
                    }
                }
            }

            private async Task ReadShutdownResponseAsync(
                long expectedRequestId,
                CancellationToken cancellationToken)
            {
                ulong jsonLengthValue =
                    await ReadUInt64Async(
                        _pythonOutput,
                        cancellationToken);

                int jsonLength =
                    ValidateAndConvertLength(
                        jsonLengthValue,
                        MaximumJsonBytes,
                        "shutdown応答JSON");

                byte[] jsonBytes =
                    new byte[jsonLength];

                await ReadExactlyAsync(
                    _pythonOutput,
                    jsonBytes,
                    0,
                    jsonBytes.Length,
                    cancellationToken);

                string responseJson =
                    Encoding.UTF8.GetString(
                        jsonBytes);

                PythonResponseMetadata metadata =
                    JsonConvert.DeserializeObject
                    <PythonResponseMetadata>(
                        responseJson);

                if (metadata == null)
                {
                    throw new InvalidDataException(
                        "shutdown応答JSONがnullです。");
                }

                ulong imageCountValue =
                    await ReadUInt64Async(
                        _pythonOutput,
                        cancellationToken);

                if (imageCountValue >
                    (ulong)MaximumResultImageCount)
                {
                    throw new InvalidDataException(
                        "shutdown応答の画像数が不正です。");
                }

                /*
                * 通常は0枚だが、
                * 画像があれば通信境界維持のため読み捨てる。
                */
                for (ulong index = 0;
                    index < imageCountValue;
                    index++)
                {
                    ulong imageLengthValue =
                        await ReadUInt64Async(
                            _pythonOutput,
                            cancellationToken);

                    int imageLength =
                        ValidateAndConvertLength(
                            imageLengthValue,
                            MaximumImageBytes,
                            "shutdown応答画像");

                    byte[] discard =
                        new byte[imageLength];

                    await ReadExactlyAsync(
                        _pythonOutput,
                        discard,
                        0,
                        discard.Length,
                        cancellationToken);
                }

                if (!metadata.Success)
                {
                    throw new PythonInferenceException(
                        metadata.Error != null
                            ? metadata.Error.Code
                            : "SHUTDOWN_ERROR",

                        metadata.Error != null
                            ? metadata.Error.Message
                            : "Python終了処理に失敗しました。",

                        metadata.RequestId);
                }

                if (!metadata.RequestId.HasValue ||
                    metadata.RequestId.Value !=
                    expectedRequestId)
                {
                    throw new InvalidDataException(
                        "shutdown応答のrequest_idが一致しません。");
                }
            }

            private async Task ReadStandardErrorLoopAsync(
                Process process)
            {
                try
                {
                    while (true)
                    {
                        string line =
                            await process.StandardError
                                .ReadLineAsync();

                        if (line == null)
                        {
                            break;
                        }

                        OnPythonLogReceived(
                            line);
                    }
                }
                catch (ObjectDisposedException)
                {
                    // 終了中
                }
                catch (InvalidOperationException)
                {
                    // Process終了中
                }
                catch (IOException)
                {
                    // stderrが閉じられた
                }
            }

            private static async Task WriteUInt64Async(
                Stream stream,
                ulong value,
                CancellationToken cancellationToken)
            {
                byte[] buffer =
                    BitConverter.GetBytes(
                        value);

                /*
                * 通信仕様はリトルエンディアン。
                */
                if (!BitConverter.IsLittleEndian)
                {
                    Array.Reverse(
                        buffer);
                }

                await stream.WriteAsync(
                    buffer,
                    0,
                    buffer.Length,
                    cancellationToken);
            }

            private static async Task<ulong>
                ReadUInt64Async(
                    Stream stream,
                    CancellationToken cancellationToken)
            {
                byte[] buffer =
                    new byte[8];

                await ReadExactlyAsync(
                    stream,
                    buffer,
                    0,
                    buffer.Length,
                    cancellationToken);

                if (!BitConverter.IsLittleEndian)
                {
                    Array.Reverse(
                        buffer);
                }

                return BitConverter.ToUInt64(
                    buffer,
                    0);
            }

            /// <summary>
            /// 指定バイト数を必ず読み切る。
            /// Stream.ReadAsyncは要求数未満しか返さない場合がある。
            /// </summary>
            private static async Task ReadExactlyAsync(
                Stream stream,
                byte[] buffer,
                int offset,
                int count,
                CancellationToken cancellationToken)
            {
                int totalRead = 0;

                while (totalRead < count)
                {
                    int read =
                        await stream.ReadAsync(
                            buffer,
                            offset + totalRead,
                            count - totalRead,
                            cancellationToken);

                    if (read == 0)
                    {
                        throw new EndOfStreamException(
                            "Python標準出力が途中で閉じられました。"
                            + " Expected="
                            + count
                            + ", Actual="
                            + totalRead);
                    }

                    totalRead += read;
                }
            }

            private static int ValidateAndConvertLength(
                ulong value,
                long maximum,
                string dataName)
            {
                if (value > (ulong)maximum)
                {
                    throw new InvalidDataException(
                        dataName
                        + "が上限を超えています。"
                        + " Length="
                        + value
                        + ", Maximum="
                        + maximum);
                }

                if (value > int.MaxValue)
                {
                    throw new InvalidDataException(
                        dataName
                        + "が.NET配列の上限を超えています。"
                        + " Length="
                        + value);
                }

                return checked((int)value);
            }

            private static async Task<bool>
                WaitForExitWithTimeoutAsync(
                    Process process,
                    TimeSpan timeout)
            {
                TaskCompletionSource<bool> completionSource =
                    new TaskCompletionSource<bool>();

                EventHandler exitedHandler = null;

                exitedHandler =
                    delegate
                    {
                        completionSource.TrySetResult(
                            true);
                    };

                process.EnableRaisingEvents = true;
                process.Exited += exitedHandler;

                try
                {
                    if (process.HasExited)
                    {
                        return true;
                    }

                    Task timeoutTask =
                        Task.Delay(
                            timeout);

                    Task completedTask =
                        await Task.WhenAny(
                            completionSource.Task,
                            timeoutTask);

                    return completedTask ==
                        completionSource.Task;
                }
                finally
                {
                    process.Exited -=
                        exitedHandler;
                }
            }

            private void ValidatePaths()
            {
                if (!File.Exists(
                        _pythonExecutablePath))
                {
                    throw new FileNotFoundException(
                        "Python実行ファイルが見つかりません。",
                        _pythonExecutablePath);
                }

                if (!File.Exists(
                        _communicationMainPath))
                {
                    throw new FileNotFoundException(
                        "communication_main.pyが見つかりません。",
                        _communicationMainPath);
                }

                if (!Directory.Exists(
                        _workingDirectory))
                {
                    throw new DirectoryNotFoundException(
                        "Python作業フォルダーが見つかりません: "
                        + _workingDirectory);
                }
            }

            private void EnsureRunning()
            {
                ThrowIfDisposed();

                if (_process == null)
                {
                    throw new InvalidOperationException(
                        "Pythonプロセスが起動されていません。");
                }

                if (_process.HasExited)
                {
                    throw new InvalidOperationException(
                        "Pythonプロセスは終了しています。"
                        + " ExitCode="
                        + _process.ExitCode);
                }

                if (_pythonInput == null ||
                    _pythonOutput == null)
                {
                    throw new InvalidOperationException(
                        "Python標準入出力が初期化されていません。");
                }
            }

            private void ThrowIfDisposed()
            {
                if (_disposed)
                {
                    throw new ObjectDisposedException(
                        GetType().FullName);
                }
            }

            private void Process_Exited(
                object sender,
                EventArgs e)
            {
                int? exitCode = null;

                try
                {
                    if (_process != null &&
                        _process.HasExited)
                    {
                        exitCode =
                            _process.ExitCode;
                    }
                }
                catch
                {
                    // 終了中
                }

                OnPythonLogReceived(
                    "Pythonプロセスが終了しました。"
                    + " ExitCode="
                    + (exitCode.HasValue
                        ? exitCode.Value.ToString()
                        : "(unknown)"));

                EventHandler handler =
                    PythonProcessExited;

                if (handler != null)
                {
                    handler(
                        this,
                        EventArgs.Empty);
                }
            }

            private void OnPythonLogReceived(
                string message)
            {
                EventHandler<string> handler =
                    PythonLogReceived;

                if (handler != null)
                {
                    handler(
                        this,
                        message);
                }
            }

            public void Dispose()
            {
                if (_disposed)
                {
                    return;
                }

                try
                {
                    /*
                    * Disposeは同期メソッドなので、
                    * ここでは短時間だけ終了を待つ。
                    *
                    * 通常はフォーム終了前に
                    * ShutdownAsync()を明示的に呼ぶ。
                    */
                    if (IsRunning)
                    {
                        Task shutdownTask =
                            ShutdownAsync();

                        shutdownTask.Wait(
                            TimeSpan.FromSeconds(7));
                    }
                }
                catch
                {
                    // Dispose処理を継続する
                }

                if (_process != null)
                {
                    try
                    {
                        if (!_process.HasExited)
                        {
                            _process.Kill();
                            _process.WaitForExit();
                        }
                    }
                    catch
                    {
                        // 終了処理を継続
                    }

                    _process.Exited -=
                        Process_Exited;

                    _process.Dispose();
                    _process = null;
                }

                if (_pythonInput != null)
                {
                    _pythonInput.Dispose();
                    _pythonInput = null;
                }

                if (_pythonOutput != null)
                {
                    _pythonOutput.Dispose();
                    _pythonOutput = null;
                }

                _communicationLock.Dispose();

                _disposed = true;
            }

    using System;
    using System.Drawing;
    using System.Drawing.Imaging;
    using System.IO;

    namespace ImageRecognition.PythonCommunication
    {
        public static class ImageByteConverter
        {
            /// <summary>
            /// BitmapをPNGファイル形式のbyte[]へ変換する。
            /// </summary>
            public static byte[] BitmapToPngBytes(
                Bitmap bitmap)
            {
                if (bitmap == null)
                {
                    throw new ArgumentNullException(
                        "bitmap");
                }

                using (MemoryStream stream =
                    new MemoryStream())
                {
                    bitmap.Save(
                        stream,
                        ImageFormat.Png);

                    return stream.ToArray();
                }
            }

            /// <summary>
            /// ImageをPNGファイル形式のbyte[]へ変換する。
            /// </summary>
            public static byte[] ImageToPngBytes(
                Image image)
            {
                if (image == null)
                {
                    throw new ArgumentNullException(
                        "image");
                }

                using (MemoryStream stream =
                    new MemoryStream())
                {
                    image.Save(
                        stream,
                        ImageFormat.Png);

                    return stream.ToArray();
                }
            }

            /// <summary>
            /// Pythonから受信したPNG byte[]をBitmapへ変換する。
            /// </summary>
            public static Bitmap PngBytesToBitmap(
                byte[] pngBytes)
            {
                if (pngBytes == null)
                {
                    throw new ArgumentNullException(
                        "pngBytes");
                }

                if (pngBytes.Length == 0)
                {
                    throw new ArgumentException(
                        "PNGバイト列が空です。",
                        "pngBytes");
                }

                using (MemoryStream stream =
                    new MemoryStream(
                        pngBytes,
                        false))
                using (Bitmap temporaryBitmap =
                    new Bitmap(stream))
                {
                    /*
                    * Bitmapは元Streamへ依存する場合があるため、
                    * Streamを閉じる前に独立したBitmapへ複製する。
                    */
                    return new Bitmap(
                        temporaryBitmap);
                }
            }
        }
    }

            /// <summary>
            /// .NET Framework 4.8用の
            /// File.ReadAllBytesAsync相当。
            /// </summary>
            private static async Task<byte[]>
                ReadAllBytesAsync(
                    string filePath,
                    CancellationToken cancellationToken)
            {
                using (FileStream stream =
                    new FileStream(
                        filePath,
                        FileMode.Open,
                        FileAccess.Read,
                        FileShare.Read,
                        81920,
                        true))
                {
                    if (stream.Length >
                        int.MaxValue)
                    {
                        throw new IOException(
                            "入力ファイルが大きすぎます。");
                    }

                    byte[] bytes =
                        new byte[(int)stream.Length];

                    int totalRead = 0;

                    while (totalRead <
                        bytes.Length)
                    {
                        int read =
                            await stream.ReadAsync(
                                bytes,
                                totalRead,
                                bytes.Length - totalRead,
                                cancellationToken);

                        if (read == 0)
                        {
                            throw new EndOfStreamException(
                                "ファイル読み込み中に"
                                + "予期せず終端に達しました。");
                        }

                        totalRead += read;
                    }

                    return bytes;
                }
            }
